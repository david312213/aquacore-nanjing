cmake_minimum_required(VERSION 3.0.2)
project(airseafly_real400_core)
find_package(catkin REQUIRED)
find_package(Eigen3 REQUIRED)
catkin_package(INCLUDE_DIRS include DEPENDS EIGEN3)
include_directories(include ${EIGEN3_INCLUDE_DIRS})
if(CATKIN_ENABLE_TESTING)
  # Catkin packages are linked into build_ws/src; resolve the real source tree.
  get_filename_component(AQUACORE_CORE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}" REALPATH)
  get_filename_component(AQUACORE_SOURCE_ROOT "${AQUACORE_CORE_SOURCE_DIR}" DIRECTORY)
  include_directories(BEFORE ${CMAKE_CURRENT_SOURCE_DIR}/test/stubs)
  add_executable(real400_core_test test/core_test.cpp)
  target_compile_features(real400_core_test PRIVATE cxx_std_14)
  add_test(NAME real400_core COMMAND real400_core_test)
  add_executable(real400_control_law_test test/control_law_test.cpp)
  target_compile_features(real400_control_law_test PRIVATE cxx_std_14)
  target_include_directories(real400_control_law_test PRIVATE ${AQUACORE_SOURCE_ROOT}/realflight_modules/px4ctrl/src)
  add_test(NAME real400_control_law COMMAND real400_control_law_test)
  add_executable(real400_bspline_test test/bspline_test.cpp)
  target_compile_features(real400_bspline_test PRIVATE cxx_std_14)
  target_include_directories(real400_bspline_test PRIVATE ${AQUACORE_SOURCE_ROOT}/planner/bspline_opt/src ${AQUACORE_SOURCE_ROOT}/planner/bspline_opt/include ${AQUACORE_SOURCE_ROOT}/planner/plan_manage/include)
  add_test(NAME real400_bspline COMMAND real400_bspline_test)
endif()
install(DIRECTORY include/ DESTINATION ${CATKIN_GLOBAL_INCLUDE_DESTINATION})
